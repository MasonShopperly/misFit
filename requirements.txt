# Host environment for this repository, pinned to the versions that produced every committed
# result. Python 3.11.15.
#
# WHY THIS FILE EXISTS. Each served component carries its own pinned
# `tesseract_requirements.txt`, so the containerized halves were reproducible while the host
# environment that actually ran every experiment was not pinned at all. A reader installing from an
# unpinned line got whatever was current, not what produced the numbers. These are those versions.
#
#   uv venv --python 3.11 .venv
#   uv pip install --python .venv/bin/python -r requirements.txt
#
# (plain `python -m venv .venv && .venv/bin/pip install -r requirements.txt` works too; `uv` is
# what was used here and is faster.)

# --- the four third-party packages the checking and figure code needs, and nothing else. PyYAML is
#     not imported by tracked code; the runtime parses each component's `tesseract_config.yaml`.
numpy==2.4.6
matplotlib==3.11.1
PyYAML==6.0.3
tesseract-core[runtime]==1.11.0
